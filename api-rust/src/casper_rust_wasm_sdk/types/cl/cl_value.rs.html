14/// Wasm/native wrapper around [`casper_types::CLValue`].
15///
16/// Minimal surface for building [`crate::types::runtime_args::RuntimeArgs`].
17/// This wrapper currently focuses on runtime-args construction.
18#[wasm_bindgen]
19#[derive(Debug, Clone, PartialEq, Eq)]
20pub struct CLValue(_CLValue);
21
22impl CLValue {
23    pub fn from_t<T: CLTyped + ToBytes>(value: T) -> Result<Self, Box<SdkError>> {
24        _CLValue::from_t(value)
25            .map(CLValue)
26            .map_err(|err| Box::new(SdkError::from(err)))
27    }
28
29    pub fn inner(&self) -> &_CLValue {
30        &self.0
31    }
32}
