8#[derive(Debug, Clone, Serialize, Default)]
9#[wasm_bindgen]
10pub struct Path {
11    path: Vec<String>,
12}
13
14#[wasm_bindgen]
15impl Path {
16    #[wasm_bindgen(constructor)]
17    pub fn new(path: JsValue) -> Self {
18        let path_string: String = if path.is_null() {
19            String::from("")
20        } else {
21            path.as_string().unwrap_or_else(|| String::from(""))
22        };
23        Path::from(path_string)
24    }
25
26    #[cfg(target_arch = "wasm32")]
27    #[wasm_bindgen(js_name = "fromArray")]
28    pub fn from_js_array(path: JsValue) -> Self {
29        let path: Array = path.into();
30        let path: Vec<String> = path
31            .iter()
32            .map(|value| {
33                value
34                    .as_string()
35                    .unwrap_or_else(|| String::from("Invalid String"))
36            })
37            .collect();
38
39        Path { path }
40    }
41
42    #[cfg(target_arch = "wasm32")]
43    #[wasm_bindgen(js_name = "toJson")]
44    pub fn to_json(&self) -> JsValue {
45        JsValue::from_serde(&self.path).unwrap_or(JsValue::null())
46    }
47
48    #[cfg(target_arch = "wasm32")]
49    #[wasm_bindgen(js_name = "toString")]
50    pub fn to_string_js_alias(&self) -> String {
51        self.to_string()
52    }
53
54    pub fn is_empty(&self) -> bool {
55        self.path.is_empty() || self.path.iter().all(|s| s.is_empty())
56    }
57}
