187// Convert PaymentStrParams to casper_client::cli::PaymentStrParams
188pub fn payment_str_params_to_casper_client(
189    payment_params: &PaymentStrParams,
190) -> _PaymentStrParams<'_> {
191    let payment_args_simple: Vec<&str> = payment_params
192        .payment_args_simple
193        .get()
194        .map_or_else(Vec::new, |args_simple| {
195            args_simple.args().iter().map(String::as_str).collect()
196        });
197
198    if let Some(payment_hash) = payment_params.payment_hash.get() {
199        return _PaymentStrParams::with_hash(
200            payment_hash.as_str(),
201            get_str_or_default(payment_params.payment_entry_point.get()),
202            payment_args_simple,
203            get_str_or_default(payment_params.payment_args_json.get()),
204        );
205    }
206
207    if let Some(payment_path) = payment_params.payment_path.get() {
208        return _PaymentStrParams::with_path(
209            payment_path.as_str(),
210            payment_args_simple,
211            get_str_or_default(payment_params.payment_args_json.get()),
212        );
213    }
214
215    if let Some(payment_name) = payment_params.payment_name.get() {
216        return _PaymentStrParams::with_name(
217            payment_name.as_str(),
218            get_str_or_default(payment_params.payment_entry_point.get()),
219            payment_args_simple,
220            get_str_or_default(payment_params.payment_args_json.get()),
221        );
222    }
223
224    if let Some(payment_package_hash) = payment_params.payment_package_hash.get() {
225        return _PaymentStrParams::with_package_hash(
226            payment_package_hash.as_str(),
227            get_str_or_default(payment_params.payment_version.get()),
228            get_str_or_default(payment_params.payment_entry_point.get()),
229            payment_args_simple,
230            get_str_or_default(payment_params.payment_args_json.get()),
231        );
232    }
233
234    if let Some(payment_package_name) = payment_params.payment_package_name.get() {
235        return _PaymentStrParams::with_package_name(
236            payment_package_name.as_str(),
237            get_str_or_default(payment_params.payment_version.get()),
238            get_str_or_default(payment_params.payment_entry_point.get()),
239            payment_args_simple,
240            get_str_or_default(payment_params.payment_args_json.get()),
241        );
242    }
243
244    if let Some(payment_package_name) = payment_params.payment_entry_point.get() {
245        return _PaymentStrParams::with_package_name(
246            payment_package_name.as_str(),
247            get_str_or_default(payment_params.payment_version.get()),
248            get_str_or_default(payment_params.payment_entry_point.get()),
249            payment_args_simple,
250            get_str_or_default(payment_params.payment_args_json.get()),
251        );
252    }
253
254    // Default to the Payment amount
255    _PaymentStrParams::with_amount(get_str_or_default(payment_params.payment_amount.get()))
256}
