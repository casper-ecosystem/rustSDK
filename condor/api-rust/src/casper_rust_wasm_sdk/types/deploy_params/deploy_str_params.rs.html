1use crate::helpers::get_current_timestamp;

8#[wasm_bindgen]
9#[derive(Debug, Clone)]
10pub struct DeployStrParams {
11    secret_key: OnceCell<String>,
12    timestamp: OnceCell<String>,
13    ttl: OnceCell<String>,
14    chain_name: OnceCell<String>,
15    session_account: OnceCell<String>,
16    gas_price_tolerance: OnceCell<String>,

18
19impl Default for DeployStrParams {
20    fn default() -> Self {
21        DeployStrParams {
22            secret_key: OnceCell::new(),
23            timestamp: OnceCell::new(),
24            ttl: OnceCell::new(),
25            chain_name: OnceCell::new(),
26            session_account: OnceCell::new(),
27            gas_price_tolerance: OnceCell::new(),
28        }
29    }
30}
31
32#[wasm_bindgen]
33impl DeployStrParams {
34    #[wasm_bindgen(constructor)]
35    pub fn new(
36        chain_name: &str,
37        session_account: &str,
38        secret_key: Option<String>,
39        timestamp: Option<String>,
40        ttl: Option<String>,
41        gas_price_tolerance: Option<String>,
42    ) -> Self {
43        let deploy_params = DeployStrParams::default();
44        deploy_params.set_chain_name(chain_name);
45        deploy_params.set_session_account(session_account);
46        if let Some(secret_key) = secret_key {
47            deploy_params.set_secret_key(&secret_key);
48        };
49        deploy_params.set_timestamp(timestamp);
50        deploy_params.set_ttl(ttl);
51        if let Some(gas_price_tolerance) = gas_price_tolerance {
52            deploy_params.set_gas_price_tolerance(gas_price_tolerance);
53        };
54        if deploy_params.gas_price_tolerance.get().is_none() {
55            deploy_params.set_gas_price_tolerance("1".to_string());
56        }
57        deploy_params
58    }
59
60    // Getter and setter for secret_key field
61    #[wasm_bindgen(getter)]
62    pub fn secret_key(&self) -> Option<String> {

64    }
65
66    #[wasm_bindgen(setter)]
67    pub fn set_secret_key(&self, secret_key: &str) {

70
71    // Getter and setter for timestamp field
72    #[wasm_bindgen(getter)]
73    pub fn timestamp(&self) -> Option<String> {
74        self.timestamp.get().cloned()
75    }
76
77    #[wasm_bindgen(setter)]
78    pub fn set_timestamp(&self, timestamp: Option<String>) {
79        if let Some(mut timestamp) = timestamp {
80            if timestamp.is_empty() {
81                timestamp = get_current_timestamp(None);
82            }
83            self.timestamp.set(timestamp.to_string()).unwrap();
84        } else {
85            let timestamp = get_current_timestamp(timestamp);
86            self.timestamp.set(timestamp).unwrap();
87        };
88    }
89
90    #[wasm_bindgen(js_name = "setDefaultTimestamp")]
91    pub fn set_default_timestamp(&self) {
92        let current_timestamp = get_current_timestamp(None);
93        self.timestamp.set(current_timestamp).unwrap();
94    }
95
96    // Getter and setter for ttl field
97    #[wasm_bindgen(getter)]
98    pub fn ttl(&self) -> Option<String> {
99        self.ttl.get().cloned()
100    }
101
102    #[wasm_bindgen(setter)]
103    pub fn set_ttl(&self, ttl: Option<String>) {
104        if let Some(mut ttl) = ttl {
105            if ttl.is_empty() {
106                ttl = get_ttl_or_default(None);
107            }
108            self.ttl.set(ttl.to_string()).unwrap();
109        } else {
110            let ttl = get_ttl_or_default(ttl.as_deref());
111            self.ttl.set(ttl).unwrap();
112        };
113    }
114
115    #[wasm_bindgen(js_name = "setDefaultTTL")]
116    pub fn set_default_ttl(&self) {
117        let ttl = get_ttl_or_default(None);
118        self.ttl.set(ttl).unwrap();
119    }
120
121    // Getter and setter for chain_name field
122    #[wasm_bindgen(getter)]
123    pub fn chain_name(&self) -> Option<String> {

125    }
126
127    #[wasm_bindgen(setter)]
128    pub fn set_chain_name(&self, chain_name: &str) {
129        self.chain_name.set(chain_name.to_string()).unwrap();
130    }
131
132    // Getter and setter for session_account field

135        self.session_account.get().cloned()

148        self.gas_price_tolerance.get().cloned()
149    }
150

153        self.gas_price_tolerance.set(gas_price_tolerance).unwrap();

157// Convert DeployStrParams to casper_client::cli::DeployStrParams
158pub fn deploy_str_params_to_casper_client(deploy_params: &DeployStrParams) -> _DeployStrParams<'_> {
159    _DeployStrParams {
160        secret_key: get_str_or_default(deploy_params.secret_key.get()),
161        timestamp: get_str_or_default(deploy_params.timestamp.get()),
162        ttl: get_str_or_default(deploy_params.ttl.get()),
163        chain_name: get_str_or_default(deploy_params.chain_name.get()),
164        session_account: get_str_or_default(deploy_params.session_account.get()),
165        gas_price_tolerance: get_str_or_default(deploy_params.gas_price_tolerance.get()),
166    }
167}

182        ttl.set("10".to_string()).unwrap();
183
184        let chain_name = OnceCell::new();
185        chain_name.set("test_chain".to_string()).unwrap();
186
187        let session_account = OnceCell::new();
188        session_account.set("account_id".to_string()).unwrap();
189
190        let gas_price_tolerance = OnceCell::new();
191        gas_price_tolerance.set("1".to_string()).unwrap();
192
193        let deploy_params = DeployStrParams {
194            secret_key,
195            timestamp,
196            ttl,
197            chain_name,
198            session_account,
199            gas_price_tolerance,
200        };
201
202        let result = deploy_str_params_to_casper_client(&deploy_params);
