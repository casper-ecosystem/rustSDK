21#[cfg(test)]
22pub async fn install_cep78() -> String {
23    use sdk_tests::{
24        config::WASM_PATH,
25        tests::helpers::{
26            get_contract_cep78_hash_keys, get_network_constants, get_user_secret_key,
27            install_cep78_if_needed,
28        },
29    };
30
31    let secret_key = get_user_secret_key(None).unwrap();
32    let account = public_key_from_secret_key(&secret_key).unwrap();
33    let public_key = PublicKey::new(&account).unwrap();
34    let account_hash = public_key.to_account_hash().to_formatted_string();
35    let (rpc_address, events_address, _, _, chain_name) = get_network_constants();
36    install_cep78_if_needed(
37        &account,
38        &secret_key,
39        Some(WASM_PATH),
40        (&rpc_address, &events_address, &chain_name),
41    )
42    .await;
43    let (contract_cep78_key, _contract_cep78_package_hash) =
44        get_contract_cep78_hash_keys(&account_hash, &rpc_address).await;
45    contract_cep78_key
46}

51#[cfg(test)]
52pub async fn get_dictionary_item(as_params: bool) -> DictionaryItemInput {
53    use sdk_tests::tests::helpers::{get_network_constants, get_user_secret_key};
54
55    let (rpc_address, events_address, _, _, chain_name) = get_network_constants();
56    let mut contract_key = CONTRACT_CEP78_KEY.lock().await; // Use the async lock
57
58    if contract_key.is_none() {
59        let contract_cep78_key = install_cep78().await;
60        let secret_key = get_user_secret_key(None).unwrap();
61        let account = public_key_from_secret_key(&secret_key).unwrap();
62        let public_key = PublicKey::new(&account).unwrap();
63        let account_hash = public_key.to_account_hash().to_formatted_string();
64
65        // Release the lock here before awaiting
66        drop(contract_key);
67
68        mint_nft(
69            &contract_cep78_key,
70            &account,
71            &account_hash,
72            &secret_key,
73            (&rpc_address, &events_address, &chain_name),
74        )
75        .await;
76
77        // Reacquire the lock
78        contract_key = CONTRACT_CEP78_KEY.lock().await;
79        *contract_key = Some(contract_cep78_key);
80    }
81
82    if as_params {
83        get_dictionary_item_params_input(contract_key.as_ref().unwrap()).await
84    } else {
85        get_dictionary_item_input(contract_key.as_ref().unwrap()).await
86    }
87}
