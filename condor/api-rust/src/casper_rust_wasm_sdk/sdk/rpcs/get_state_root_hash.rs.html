184impl SDK {
185    /// Retrieves state root hash information based on the provided options.
186    ///
187    /// # Arguments
188    ///
189    /// * `maybe_block_identifier` - An optional `BlockIdentifierInput` for specifying a block identifier.
190    /// * `verbosity` - An optional `Verbosity` level for controlling the output verbosity.
191    /// * `rpc_address` - An optional string specifying the rpc address to use for the request.
192    ///
193    /// # Returns
194    ///
195    /// A `Result` containing either a `_GetStateRootHashResult` or a `SdkError` in case of an error.
196    ///
197    /// # Errors
198    ///
199    /// Returns a `SdkError` if there is an error during the retrieval process.
200    pub async fn get_state_root_hash(
201        &self,
202        maybe_block_identifier: Option<BlockIdentifierInput>,
203        verbosity: Option<Verbosity>,
204        rpc_address: Option<String>,
205    ) -> Result<SuccessResponse<_GetStateRootHashResult>, SdkError> {
206        //log("get_state_root_hash!");
207        let random_id = rand::rng().random::<u64>().to_string();
208
209        if let Some(BlockIdentifierInput::String(maybe_block_id)) = maybe_block_identifier {
210            get_state_root_hash_cli(
211                &random_id,
212                &self.get_rpc_address(rpc_address),
213                self.get_verbosity(verbosity).into(),
214                &maybe_block_id,
215            )
216            .await
217            .map_err(SdkError::from)
218        } else {
219            let maybe_block_identifier =
220                if let Some(BlockIdentifierInput::BlockIdentifier(maybe_block_identifier)) =
221                    maybe_block_identifier
222                {
223                    Some(maybe_block_identifier)
224                } else {
225                    None
226                };
227
228            get_state_root_hash_lib(
229                JsonRpcId::from(random_id),
230                &self.get_rpc_address(rpc_address),
231                self.get_verbosity(verbosity).into(),
232                maybe_block_identifier.map(Into::into),
233            )
234            .await
235            .map_err(SdkError::from)
236        }
237    }
238}
