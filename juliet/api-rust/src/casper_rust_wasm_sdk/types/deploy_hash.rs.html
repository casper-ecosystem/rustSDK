13#[derive(Debug, Deserialize, Clone, Serialize)]
14#[wasm_bindgen]
15pub struct DeployHash(_DeployHash);

17impl DeployHash {
18    pub fn new(deploy_hash_hex_str: &str) -> Result<DeployHash, Box<SdkError>> {
19        let bytes =
20            hex::decode(deploy_hash_hex_str).map_err(|err| SdkError::FailedToDecodeHex {
21                context: "DeployHash::new",
22                error: format!("Decoding hex string {:?}", err),
23            })?;
24        let mut hash = [0u8; _Digest::LENGTH];
25        hash.copy_from_slice(&bytes);
26        Self::from_digest(Digest::from(hash))
27    }
28
29    pub fn from_digest(digest: Digest) -> Result<DeployHash, Box<SdkError>> {
30        Ok(_DeployHash::new(digest.value()).into())
31    }
32}

82// Both Node and Client expose DeployHash but differently
83impl From<DeployHash> for _DeployHashClient {
84    fn from(deploy_hash: DeployHash) -> Self {
85        let mut bytes: [u8; DEPLOY_HASH_LENGTH] = [0; DEPLOY_HASH_LENGTH];
86        bytes.copy_from_slice(deploy_hash.0.as_ref());
87        let digest = Digest::from_raw(bytes.to_vec()).unwrap();
88        _DeployHashClient::new(digest.into())
89    }
90}

92impl From<_DeployHashClient> for DeployHash {
93    fn from(deploy_hash: _DeployHashClient) -> Self {
94        let digest = deploy_hash.inner();
95        let deploy_hash = _DeployHash::new(digest.into());
96        DeployHash(deploy_hash)
97    }
