44/// Gets the current timestamp.
45///
46/// # Arguments
47///
48/// * `timestamp` - An optional timestamp value in milliseconds since the Unix epoch.
49///
50/// # Returns
51///
52/// A string containing the current timestamp in RFC3339 format.
53pub fn get_current_timestamp(timestamp: Option<String>) -> String {
54    let parsed_timestamp = timestamp.as_ref().and_then(|ts| ts.parse::<i64>().ok());
55    let current_timestamp = parsed_timestamp
56        .map(|parsed_time| DateTime::from_timestamp(parsed_time / 1000, 0).unwrap_or_else(Utc::now))
57        .unwrap_or_else(Utc::now);
58    current_timestamp.to_rfc3339_opts(SecondsFormat::Secs, true)
59}

61/// Computes the Blake2b hash of the provided metadata.
62///
63/// # Arguments
64///
65/// * `meta_data` - A reference to a string containing the metadata to be hashed.
66///
67/// # Returns
68///
69/// A hexadecimal string representing the Blake2b hash of the input metadata.
70pub fn get_blake2b_hash(meta_data: &str) -> String {
71    let mut result = [0; BLAKE2B_DIGEST_LENGTH];
72    let mut hasher = VarBlake2b::new(BLAKE2B_DIGEST_LENGTH).expect("should create hasher");
73
74    hasher.update(meta_data);
75    hasher.finalize_variable(|slice| {
76        result.copy_from_slice(slice);
77    });
78    hex::encode(result).to_lowercase()
79}

81/// Creates a dictionary item key by concatenating the serialized bytes of the key and value.
82///
83/// # Arguments
84///
85/// * `key` - The key to be serialized.
86/// * `value` - The value to be serialized.
87///
88/// # Returns
89///
90/// A hexadecimal-encoded string representing the dictionary item key.
91///
92/// # Panics
93///
94/// Panics if the hasher cannot be created.
95///
96/// # Example
97///
98/// ```rust
99/// use casper_types::{U256, U512};
100/// use casper_rust_wasm_sdk::helpers::make_dictionary_item_key;
101/// use casper_rust_wasm_sdk::types::key::Key;
102/// // Key / Value as U256
103/// let key = Key::from_formatted_str( "account-hash-e11bfffe63bf899ea07117af8a2bb43ef0078c0e38ebee6b6cb0b0e39c233538").unwrap();
104/// let value = U256::from(1);
105/// let dictionary_item_key = make_dictionary_item_key(&key, &value);
106/// println!("Dictionary Item Key (Key/Value as U256): {}", dictionary_item_key);
107/// assert_eq!(dictionary_item_key,"145f6211a24c0a8af16b47e7aa58431ea25172eb402903b3c25ac92b9784c7a9".to_string());
108/// // Key / Value as Key
109/// let key = Key::from_formatted_str( "account-hash-813428ce1a9805f1087db07e6017c6c4f5af0ee78a05591bb6577763e89b4f1f").unwrap();
110/// let value = Key::from_formatted_str("account-hash-e11bfffe63bf899ea07117af8a2bb43ef0078c0e38ebee6b6cb0b0e39c233538").unwrap();
111/// let dictionary_item_key = make_dictionary_item_key(&key, &value);
112/// println!("Dictionary Item Key (Key/Value as Key): {}", dictionary_item_key);
113/// assert_eq!(dictionary_item_key,"1e26dc82db208943c3785c0e11b9d78b9c408fee748c78dda5a5d016840dedca".to_string());
114/// ```
115pub fn make_dictionary_item_key<V: ToBytes>(key: &Key, value: &V) -> String {
116    let key: _Key = _Key::from(key.clone());
117    let mut bytes_a = key.to_bytes().unwrap_or_default();
118    let mut bytes_b = value.to_bytes().unwrap_or_default();
119
120    bytes_a.append(&mut bytes_b);
121
122    let mut result = [0; BLAKE2B_DIGEST_LENGTH];
123    let mut hasher = VarBlake2b::new(BLAKE2B_DIGEST_LENGTH).expect("should create hasher");
124
125    hasher.update(bytes_a);
126    hasher.finalize_variable(|slice| {
127        result.copy_from_slice(slice);
128    });
129    hex::encode(result)
130}

132/// Convert a formatted account hash to a base64-encoded string (cep-18 key encoding).
133///
134/// # Arguments
135///
136/// * `formatted_account_hash` - A hex-formatted string representing the account hash.
137///
138/// Example: "account-hash-b485c074cef7ccaccd0302949d2043ab7133abdb14cfa87e8392945c0bd80a5f"
139///
140/// # Returns
141///
142/// Returns a `Result` with the base64-encoded string on success, or a `FromStrError` on failure.
143/// Example: "ALSFwHTO98yszQMClJ0gQ6txM6vbFM+ofoOSlFwL2Apf"
144pub fn get_base64_key_from_account_hash(
145    formatted_account_hash: &str,
146) -> Result<String, Box<SdkError>> {
147    let account_hash = AccountHash::from_formatted_str(formatted_account_hash)?;
148    let key = Key::from_account(account_hash).to_bytes().unwrap();
149    Ok(general_purpose::STANDARD.encode(key)) // base64.encode
150}

331/// Converts a hexadecimal string to a vector of unsigned 8-bit integers (Uint8Array).
332///
333/// # Arguments
334///
335/// * `hex_string` - The hexadecimal string to convert.
336///
337/// # Returns
338///
339/// A vector of unsigned 8-bit integers (Uint8Array) containing the converted value.
340pub fn hex_to_uint8_vec(hex_string: &str) -> Vec<u8> {
341    let mut bytes = Vec::with_capacity(hex_string.len() / 2);
342    let mut hex_chars = hex_string.chars();
343    while let (Some(a), Some(b)) = (hex_chars.next(), hex_chars.next()) {
344        if let Ok(byte) = u8::from_str_radix(&format!("{a}{b}"), 16) {
345            bytes.push(byte);
346        } else {
347            // If an invalid hex pair is encountered, return an empty vector.
348            return Vec::new();
349        }
350    }
351    bytes
352}

355///
356/// # Arguments
357///
358/// * `hex_string` - The hexadecimal string to convert.
359///
360/// # Returns
361///
362/// A regular string containing the converted value.
363pub fn hex_to_string(hex_string: &str) -> String {
364    match hex::decode(hex_string) {
365        Ok(bytes) => String::from_utf8_lossy(&bytes).to_string(),
366        Err(_) => hex_string.to_string(),
367    }

370/// Converts motes to CSPR (Casper tokens).
371///
372/// # Arguments
373///
374/// * `motes` - The motes value to convert.
375///
376/// # Returns
377///
378/// A string representing the CSPR amount.
379pub fn motes_to_cspr(motes: &str) -> Result<String, Box<SdkError>> {
380    match BigDecimal::from_str(motes) {
381        Ok(motes_decimal) => {
382            let divisor = BigDecimal::from(1_000_000_000);
383            let cspr_decimal = &motes_decimal / divisor;
384            let formatted_cspr = format!("{cspr_decimal:.2}");
385
386            if formatted_cspr.ends_with(".00") {
387                Ok(formatted_cspr.replace(".00", ""))
388            } else {
389                Ok(formatted_cspr)
390            }
391        }
392        Err(err) => Err(Box::new(SdkError::CustomError {
393            context: "Failed to parse input as BigDecimal",
394            error: format!("{err:?}"),
395        })),
396    }
397}

399/// Pretty prints a serializable value as a JSON string.
400///
401/// # Arguments
402///
403/// * `value` - The serializable value to pretty print.
404/// * `verbosity` - An optional verbosity level for pretty printing.
405///
406/// # Returns
407///
408/// A JSON string representing the pretty printed value.
409pub fn json_pretty_print<T>(value: T, verbosity: Option<Verbosity>) -> Result<String, Box<SdkError>>
410where
411    T: Serialize,
412{
413    let deserialized = serde_json::to_value(&value).map_err(SdkError::from)?;
414
415    match verbosity {
416        Some(Verbosity::Low) | None => Ok(deserialized.to_string()),
417        Some(Verbosity::Medium) => casper_types::json_pretty_print(&deserialized).map_err(|err| {
418            Box::new(SdkError::CustomError {
419                context: "Error in json_pretty_print",
420                error: format!("{err}"),
421            })
422        }),
423        Some(Verbosity::High) => {
424            serde_json::to_string_pretty(&deserialized).map_err(|err| Box::new(SdkError::from(err)))
425        }
426    }
427}

429/// Inserts a JavaScript value argument into a RuntimeArgs map.
430///
431/// # Arguments
432///
433/// * `args` - The RuntimeArgs map to insert the argument into.
434/// * `js_value_arg` - The JavaScript value argument to insert.
435///
436/// # Returns
437///
438/// The modified `RuntimeArgs` map.
439#[cfg(target_arch = "wasm32")]
440pub fn insert_js_value_arg(
441    args: &mut RuntimeArgs,
442    js_value_arg: JsValue,
443) -> Result<&RuntimeArgs, Box<SdkError>> {
444    if js_sys::Object::instanceof(&js_value_arg) {
445        let json_arg: JsonArg = js_value_arg
446            .into_serde()
447            .map_err(|err| SdkError::CustomError {
448                context: "Error converting to JsonArg",
449                error: format!("{err:?}"),
450            })?;
451
452        let named_arg = NamedArg::try_from(json_arg).map_err(|err| {
453            Box::new(SdkError::CustomError {
454                context: "Error converting to NamedArg",
455                error: format!("{err:?}"),
456            })
457        })?;
458
459        args.insert_cl_value(named_arg.name(), named_arg.cl_value().clone());
460    } else if let Some(string_arg) = js_value_arg.as_string() {
461        let simple_arg = string_arg;
462        casper_client::cli::insert_arg(&simple_arg, args).map_err(|err| {
463            Box::new(SdkError::CustomError {
464                context: "Error inserting simple arg",
465                error: format!("{err:?}"),
466            })
467        })?;
468    } else {
469        return Err(Box::new(SdkError::CustomError {
470            context: "Error converting to JsonArg or Simple Arg",
471            error: String::from("Unknown argument type"),
472        }));
473    }
474
475    Ok(args)
476}

478/// Inserts an argument into a RuntimeArgs map.
479///
480/// # Arguments
481///
482/// * `args` - The RuntimeArgs map to insert the argument into.
483/// * `new_arg` - The argument as a string.
484///
485/// # Returns
486///
487/// The modified `RuntimeArgs` map.
488pub(crate) fn insert_arg(args: &mut RuntimeArgs, new_arg: String) -> &RuntimeArgs {
489    match serde_json::from_str::<JsonArg>(&new_arg) {
490        Ok(json_arg) => {
491            if let Ok(named_arg) = NamedArg::try_from(json_arg.clone()) {
492                // JSON args
493                args.insert_cl_value(named_arg.name(), named_arg.cl_value().clone());
494            }
495        }
496        Err(_) => {
497            // Simple args
498            let _ = casper_client::cli::insert_arg(&new_arg, args);
499        }
500    }
501    args
502}

673        struct TestData {
674            age: i32,
675            name: String,
676        }
677
678        let data = TestData {
679            age: 42,
680            name: "Joe".to_string(),
681        };
682
683        let result = json_pretty_print(data.clone(), None).unwrap();

685
686        let result = json_pretty_print(data.clone(), Some(Verbosity::Low)).unwrap();
687        assert_eq!(result, "{\"age\":42,\"name\":\"Joe\"}");
688
689        let result = json_pretty_print(data.clone(), Some(Verbosity::Medium)).unwrap();
690        assert_eq!(result, "{\n  \"age\": 42,\n  \"name\": \"Joe\"\n}");
691
692        let result = json_pretty_print(data, Some(Verbosity::High)).unwrap();
693        assert_eq!(result, "{\n  \"age\": 42,\n  \"name\": \"Joe\"\n}");
694    }
