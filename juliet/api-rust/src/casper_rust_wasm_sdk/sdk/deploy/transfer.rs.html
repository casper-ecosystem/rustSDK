25    ///
26    /// # Arguments
27    ///
28    /// * `amount` - The amount to transfer.
29    /// * `target_account` - The target account.
30    /// * `transfer_id` - An optional transfer ID (defaults to a random number).
31    /// * `deploy_params` - The deployment parameters.
32    /// * `payment_params` - The payment parameters.
33    /// * `verbosity` - The verbosity level for logging (optional).
34    /// * `node_address` - The address of the node to connect to (optional).
35    ///
36    /// # Returns
37    ///
38    /// A `Result` containing the result of the transfer or a `JsError` in case of an error.

41    pub async fn transfer_js_alias(
42        &self,
43        amount: &str,
44        target_account: &str,
45        transfer_id: Option<String>,
46        deploy_params: DeployStrParams,
47        payment_params: PaymentStrParams,
48        verbosity: Option<Verbosity>,
49        node_address: Option<String>,
50    ) -> Result<PutDeployResult, JsError> {
51        let result = self
52            .transfer(
53                amount,
54                target_account,
55                transfer_id,
56                deploy_params,
57                payment_params,
58                verbosity,
59                node_address,
60            )
61            .await;
62        match result {
63            Ok(data) => Ok(data.result.into()),
64            Err(err) => {
65                let err = &format!("Error occurred with {err:?}");
66                Err(JsError::new(err))
67            }
68        }

72impl SDK {
73    /// Perform a transfer of funds.
74    ///
75    /// # Arguments
76    ///
77    /// * `amount` - The amount to transfer.
78    /// * `target_account` - The target account.
79    /// * `transfer_id` - An optional transfer ID (defaults to a random number).
80    /// * `deploy_params` - The deployment parameters.
81    /// * `payment_params` - The payment parameters.
82    /// * `verbosity` - The verbosity level for logging (optional).
83    /// * `node_address` - The address of the node to connect to (optional).
84    ///
85    /// # Returns
86    ///
87    /// A `Result` containing the result of the transfer or a `SdkError` in case of an error.
88    #[allow(clippy::too_many_arguments)]
89    pub async fn transfer(
90        &self,
91        amount: &str,
92        target_account: &str,
93        transfer_id: Option<String>,
94        deploy_params: DeployStrParams,
95        payment_params: PaymentStrParams,
96        verbosity: Option<Verbosity>,
97        node_address: Option<String>,
98    ) -> Result<SuccessResponse<_PutDeployResult>, Box<SdkError>> {
99        //log("transfer!");
100        let transfer_id = if let Some(transfer_id) = transfer_id {
101            transfer_id
102        } else {
103            rand::thread_rng().gen::<u64>().to_string()
104        };
105        let deploy = make_transfer(
106            "",
107            amount,
108            target_account,
109            &transfer_id,
110            deploy_str_params_to_casper_client(&deploy_params),
111            payment_str_params_to_casper_client(&payment_params),
112            false,
113        )
114        .map_err(|err| Box::new(SdkError::from(err)))?;
115
116        self.put_deploy(deploy.into(), verbosity, node_address)
117            .await
118            .map_err(|err| Box::new(SdkError::from(err)))
119    }
120}
